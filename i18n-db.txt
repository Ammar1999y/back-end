// Drizzle schema
export const places = pgTable('places', {
  id: serial('id').primaryKey(),
  slug: varchar('slug', { length: 200 }).unique(),

  // Global fields
  year: integer('year'),
  latitude: decimal('latitude', { precision: 10, scale: 8 }),
  longitude: decimal('longitude', { precision: 11, scale: 8 }),
  mapUrl: text('map_url'),
  cityId: integer('city_id').references(() => cities.id),
  heritageTypeId: integer('heritage_type_id').references(() => heritageTypes.id),
  eraId: integer('era_id').references(() => eras.id),
  isActive: boolean('is_active').default(true),

//  -- Default/fallback values (لجميع اللغات)
//   title TEXT NOT NULL,           -- Required (fallback)
//   content TEXT,

  // All translations
  translations: jsonb('translations').$type<PlaceTranslations>().notNull().default({}),

  createdAt: timestamp('created_at').defaultNow(),
  updatedAt: timestamp('updated_at'),
});



interface PlaceTranslation {
  title: string;
  subtitle?: string;
  shortDescription: string;
  description?: string;  // Rich text JSON
  availability?: string;
  cards?: Array<{
    title: string;
    description?: string;
    isActive: boolean;
    order: number;
  }>;
}

type PlaceTranslations = {
  [languageCode: string]: PlaceTranslation;
};
